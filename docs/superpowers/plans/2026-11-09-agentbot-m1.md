**Goal:** 构建一个可运行的纯 Python lumina，直连 SnowLuma 的 OneBot（WS 收事件 + HTTP 调动作），用 DeepSeek-flash 跑 ReAct 多步循环，带 SQLite 记忆、子进程沙盒、分级 allowlist 工具层，完全替换 AstrBot。

**Tech Stack:** Python 3.12、openai（DeepSeek 兼容）、httpx、websockets、pydantic-settings、aiosqlite、pytest + pytest-asyncio。

- [ ] **Step 1: 写 lumina/onebot/schema.py**

```python
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Literal

RiskLevel = Literal["safe", "risky", "forbidden"]


@dataclass
class ActionParam:
    name: str
    type: str
    description: str
    required: bool = False


@dataclass
class ActionSchema:
    name: str
    category: str
    description: str
    risk: RiskLevel
    params: list[ActionParam]


def load_actions(path: str | Path) -> list[ActionSchema]:
    data = json.loads(Path(path).read_text(encoding="utf-8"))
    return [
        ActionSchema(
            name=a["name"],
            category=a["category"],
            description=a["description"],
            risk=a["risk"],
            params=[ActionParam(**p) for p in a.get("params", [])],
        )
        for a in data["actions"]
    ]


def to_openai_tool(action: ActionSchema) -> dict:
    properties = {
        p.name: {"type": p.type, "description": p.description} for p in action.params
    }
    required = [p.name for p in action.params if p.required]
    return {
        "type": "function",
        "function": {
            "name": action.name,
            "description": action.description,
            "parameters": {
                "type": "object",
                "properties": properties,
                "required": required,
            },
        },
    }
```

- [ ] **Step 2: 写 lumina/tools/registry.py**

```python
from __future__ import annotations

from dataclasses import dataclass

from lumina.onebot.schema import ActionSchema, RiskLevel, to_openai_tool


@dataclass
class Tool:
    name: str
    schema: ActionSchema

    def to_openai(self) -> dict:
        return to_openai_tool(self.schema)


class ToolRegistry:
    def __init__(self, actions: list[ActionSchema], allow_risky: set[str] | None = None):
        self._by_name: dict[str, Tool] = {}
        self._allow_risky = allow_risky or set()
        for a in actions:
            if a.risk == "forbidden":
                continue
            self._by_name[a.name] = Tool(name=a.name, schema=a)

    def _enabled(self, tool: Tool) -> bool:
        if tool.schema.risk == "safe":
            return True
        return tool.name in self._allow_risky

    def openai_tools(self) -> list[dict]:
        return [t.to_openai() for t in self._by_name.values() if self._enabled(t)]

    def get(self, name: str) -> Tool | None:
        tool = self._by_name.get(name)
        if tool is None or not self._enabled(tool):
            return None
        return tool
```

```python
from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Literal


@dataclass
class ToolCall:
    id: str
    name: str
    arguments: dict[str, Any]


@dataclass
class ChatMessage:
    role: Literal["system", "user", "assistant", "tool"]
    content: str
    tool_calls: list[ToolCall] = field(default_factory=list)
    tool_call_id: str | None = None


@dataclass
class ChatResponse:
    message: ChatMessage
    finish_reason: str


class LLMProvider(ABC):
    @abstractmethod
    async def complete(
        self,
        messages: list[ChatMessage],
        tools: list[dict],
        max_tokens: int = 2048,
    ) -> ChatResponse:
        raise NotImplementedError
```

- [ ] **Step 2: 写 lumina/llm/openai_compat.py**

```python
from __future__ import annotations

import json

from openai import AsyncOpenAI

from lumina.llm.base import ChatMessage, ChatResponse, LLMProvider, ToolCall


class OpenAICompatProvider(LLMProvider):
    def __init__(self, api_key: str, base_url: str, model: str):
        self._model = model
        self._client = AsyncOpenAI(api_key=api_key, base_url=base_url)

    def _to_openai(self, m: ChatMessage) -> dict:
        msg: dict = {"role": m.role, "content": m.content}
        if m.tool_calls:
            msg["tool_calls"] = [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {"name": tc.name, "arguments": json.dumps(tc.arguments)},
                }
                for tc in m.tool_calls
            ]
        if m.tool_call_id:
            msg["tool_call_id"] = m.tool_call_id
        return msg

    async def complete(
        self,
        messages: list[ChatMessage],
        tools: list[dict],
        max_tokens: int = 2048,
    ) -> ChatResponse:
        resp = await self._client.chat.completions.create(
            model=self._model,
            messages=[self._to_openai(m) for m in messages],
            tools=tools or None,
            max_tokens=max_tokens,
        )
        choice = resp.choices[0]
        msg = choice.message
        tool_calls = [
            ToolCall(
                id=tc.id,
                name=tc.function.name,
                arguments=json.loads(tc.function.arguments or "{}"),
            )
            for tc in (msg.tool_calls or [])
        ]
        return ChatResponse(
            message=ChatMessage(
                role="assistant",
                content=msg.content or "",
                tool_calls=tool_calls,
            ),
            finish_reason=choice.finish_reason or "",
        )
```

- [ ] **Step 1: 写 lumina/onebot/client.py（含 call_action；WS 方法在 Task 5 补）**

```python
from __future__ import annotations

import httpx


class OneBotClient:
    def __init__(self, http_url: str, ws_url: str, http_token: str = "", ws_token: str = ""):
        self.http_url = http_url.rstrip("/")
        self.ws_url = ws_url
        self.http_token = http_token
        self.ws_token = ws_token
        self._http = httpx.AsyncClient(base_url=self.http_url, timeout=30.0)

    def _headers(self) -> dict:
        if self.http_token:
            return {"Authorization": f"Bearer {self.http_token}"}
        return {}

    async def call_action(self, action: str, params: dict) -> dict:
        resp = await self._http.post(f"/{action}", json=params, headers=self._headers())
        resp.raise_for_status()
        data = resp.json()
        # OneBot 约定：status == "ok" 表示成功
        if data.get("status") != "ok":
            raise RuntimeError(f"onebot action {action} failed: {data}")
        return data.get("data", data)

    async def aclose(self) -> None:
        await self._http.aclose()
```

from dataclasses import dataclass


@dataclass
class MessageEvent:
    message_type: str  # "group" | "private"
    user_id: int
    group_id: int | None
    message: list[dict]
    raw_message: str
    message_id: int


def parse_event(payload: dict) -> MessageEvent | None:
    if payload.get("post_type") != "message":
        return None
    if payload.get("message_type") not in ("group", "private"):
        return None
    return MessageEvent(
        message_type=payload["message_type"],
        user_id=int(payload["user_id"]),
        group_id=int(payload["group_id"]) if payload.get("group_id") else None,
        message=payload.get("message", []),
        raw_message=payload.get("raw_message", ""),
        message_id=int(payload["message_id"]),
    )

    async def connect_ws(self, on_event) -> None:
        import websockets

        headers = {}
        if self.ws_token:
            headers["Authorization"] = f"Bearer {self.ws_token}"
        async for ws in websockets.connect(self.ws_url, additional_headers=headers):
            try:
                async for raw in ws:
                    import json

                    event = json.loads(raw)
                    if event.get("post_type") == "message":
                        on_event(event)
            except websockets.ConnectionClosed:
                continue

```python
from __future__ import annotations

import aiosqlite

from lumina.llm.base import ChatMessage


class MemoryStore:
    def __init__(self, db_path: str):
        self._db_path = db_path

    async def _conn(self):
        conn = await aiosqlite.connect(self._db_path)
        await conn.execute(
            "CREATE TABLE IF NOT EXISTS messages ("
            "id INTEGER PRIMARY KEY AUTOINCREMENT,"
            "scope TEXT NOT NULL, role TEXT NOT NULL, content TEXT NOT NULL,"
            "ts INTEGER NOT NULL)"
        )
        await conn.commit()
        return conn

    async def append(self, scope: str, role: str, content: str) -> None:
        import time

        conn = await self._conn()
        await conn.execute(
            "INSERT INTO messages (scope, role, content, ts) VALUES (?, ?, ?, ?)",
            (scope, role, content, int(time.time())),
        )
        await conn.commit()
        await conn.close()

    async def recent(self, scope: str, limit: int = 20) -> list[ChatMessage]:
        conn = await self._conn()
        cur = await conn.execute(
            "SELECT role, content FROM messages WHERE scope = ? "
            "ORDER BY id DESC LIMIT ?",
            (scope, limit),
        )
        rows = await cur.fetchall()
        await conn.close()
        rows = list(reversed(rows))
        return [ChatMessage(role=r, content=c) for r, c in rows]

    async def clear(self, scope: str) -> None:
        conn = await self._conn()
        await conn.execute("DELETE FROM messages WHERE scope = ?", (scope,))
        await conn.commit()
        await conn.close()
```

**Interfaces:**
- Produces: `SandboxResult(stdout, stderr, returncode, timed_out)`, `run_python(code: str, timeout: float = 10.0) -> SandboxResult`（async，用 `asyncio.create_subprocess_exec` 跑 `sys.executable -c code`，独立 cwd，超时 kill）。

- [ ] **Step 1: 写 lumina/sandbox/runner.py**

```python
from __future__ import annotations

import asyncio
import sys
from dataclasses import dataclass


@dataclass
class SandboxResult:
    stdout: str
    stderr: str
    returncode: int
    timed_out: bool = False


async def run_python(code: str, timeout: float = 10.0) -> SandboxResult:
    proc = await asyncio.create_subprocess_exec(
        sys.executable,
        "-c",
        code,
        stdout=asyncio.subprocess.PIPE,
        stderr=asyncio.subprocess.PIPE,
    )
    try:
        stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=timeout)
        return SandboxResult(
            stdout=stdout.decode(errors="replace"),
            stderr=stderr.decode(errors="replace"),
            returncode=proc.returncode if proc.returncode is not None else -1,
        )
    except asyncio.TimeoutError:
        proc.kill()
        await proc.wait()
        return SandboxResult(
            stdout="", stderr="timed out", returncode=-1, timed_out=True
        )
```

```python
from __future__ import annotations

import json

from lumina.config import Settings
from lumina.llm.base import ChatMessage, LLMProvider
from lumina.memory.store import MemoryStore
from lumina.onebot.client import OneBotClient
from lumina.onebot.events import MessageEvent
from lumina.tools.registry import ToolRegistry

SYSTEM_PROMPT = (
    "你是 QQ 群助手。根据用户意图调用可用工具完成任务，"
    "观察工具返回结果后再决定下一步，直到能给出最终回复。"
    "无法完成时说明原因。回复用简洁中文，不用 emoji。"
)


class Agent:
    def __init__(
        self,
        provider: LLMProvider,
        registry: ToolRegistry,
        client: OneBotClient,
        memory: MemoryStore,
        settings: Settings,
    ):
        self.provider = provider
        self.registry = registry
        self.client = client
        self.memory = memory
        self.settings = settings

    def _scope(self, event: MessageEvent) -> str:
        return f"{event.message_type}:{event.group_id or event.user_id}"

    async def handle_message(self, event: MessageEvent) -> str:
        scope = self._scope(event)
        await self.memory.append(scope, "user", event.raw_message)

        messages = [ChatMessage(role="system", content=SYSTEM_PROMPT)]
        messages += await self.memory.recent(scope)
        tools = self.registry.openai_tools()

        for _ in range(self.settings.max_steps):
            resp = await self.provider.complete(messages, tools)
            msg = resp.message
            if not msg.tool_calls:
                await self.memory.append(scope, "assistant", msg.content)
                return msg.content or "（无回复）"

            messages.append(msg)
            for tc in msg.tool_calls:
                tool = self.registry.get(tc.name)
                if tool is None:
                    result = {"error": f"工具 {tc.name} 不可用"}
                else:
                    try:
                        result = await tool.run(tc.arguments, self.client)
                    except Exception as e:  # noqa: BLE001
                        result = {"error": str(e)}
                messages.append(
                    ChatMessage(
                        role="tool",
                        content=json.dumps(result, ensure_ascii=False)[:4000],
                        tool_call_id=tc.id,
                    )
                )

        return "（已达最大步数，未完成）"
```

from __future__ import annotations

import asyncio
from pathlib import Path

from lumina.agent.loop import Agent
from lumina.config import get_settings
from lumina.llm.openai_compat import OpenAICompatProvider
from lumina.memory.store import MemoryStore
from lumina.onebot.client import OneBotClient
from lumina.onebot.events import parse_event
from lumina.onebot.schema import load_actions
from lumina.tools.registry import ToolRegistry

ACTIONS_PATH = Path(__file__).parent / "tools" / "snowluma_actions.json"


async def main() -> None:
    settings = get_settings()
    actions = load_actions(ACTIONS_PATH)
    registry = ToolRegistry(actions, allow_risky=set(settings.allow_risky))
    provider = OpenAICompatProvider(
        api_key=settings.deepseek_api_key,
        base_url=settings.deepseek_base_url,
        model=settings.deepseek_model,
    )
    client = OneBotClient(
        http_url=settings.snowluma_http_url,
        ws_url=settings.snowluma_ws_url,
        http_token=settings.snowluma_http_token,
        ws_token=settings.snowluma_ws_token,
    )
    Path(settings.db_path).parent.mkdir(parents=True, exist_ok=True)
    memory = MemoryStore(settings.db_path)
    agent = Agent(provider, registry, client, memory, settings)

    async def on_event(payload: dict) -> None:
        event = parse_event(payload)
        if event is None:
            return
        try:
            reply = await agent.handle_message(event)
            if event.message_type == "group":
                await client.call_action(
                    "send_group_msg",
                    {"group_id": event.group_id, "message": reply},
                )
            else:
                await client.call_action(
                    "send_private_msg",
                    {"user_id": event.user_id, "message": reply},
                )
        except Exception as e:  # noqa: BLE001
            import logging

            logging.getLogger("lumina").exception("handle message failed: %s", e)

    await client.connect_ws(on_event)


if __name__ == "__main__":
    asyncio.run(main())
